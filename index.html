// Control del Botón Modo Teatro const theaterBtn = document.getElementById('theater-btn'); const mainContainer = document.querySelector('.main-container'); theaterBtn.addEventListener('click', function() { mainContainer.classList.toggle('theater-mode'); if (mainContainer.classList.contains('theater-mode')) { theaterBtn.innerText = 'Mostrar Chat'; } else { theaterBtn.innerText = 'Modo Teatro'; } });
